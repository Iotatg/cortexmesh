  add_compile_options(-mpopcnt)

# Grammar regression: the agreement rule must survive on pairs the model was not tuned on. The
# tuned set alone scored 100% while the held-out set scored 57.9% (D60), so both are gated here.
set(GRAMMAR_ARGS ${CMAKE_SOURCE_DIR}/data/agree_train.txt 10000 3000000 1,2,3,4,5,6 0.0001 1 0.5 10 0 48 0.9 4 8 0.05 1 0 1.0 1.0 1 4 0.25)
add_test(NAME grammar_tuned COMMAND grammar ${GRAMMAR_ARGS})
set_tests_properties(grammar_tuned PROPERTIES PASS_REGULAR_EXPRESSION "OVERALL +20 +20 +100\\.0%")
add_test(NAME grammar_heldout COMMAND grammar ${GRAMMAR_ARGS} ${CMAKE_SOURCE_DIR}/data/agree_heldout.txt)
set_tests_properties(grammar_heldout PROPERTIES PASS_REGULAR_EXPRESSION "OVERALL +19 +19 +100\\.0%")
add_test(NAME grammar_hard COMMAND grammar ${GRAMMAR_ARGS} ${CMAKE_SOURCE_DIR}/data/agree_hard.txt)
set_tests_properties(grammar_hard PROPERTIES PASS_REGULAR_EXPRESSION "OVERALL +19 +19 +100\\.0%")
